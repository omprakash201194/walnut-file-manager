```tsx
// src/renderer/src/components/ScanProgress.tsx
import { Progress } from '@/components/ui/progress'
import { Button } from '@/components/ui/button'
import { formatBytes } from '@/lib/utils'
import type { ScanProgressEvent } from '../../../shared/ipc-types'

interface ScanProgressProps {
  folderPath: string
  progress: ScanProgressEvent | null
  onCancel: () => void
}

function ScanProgress({ folderPath, progress, onCancel }: ScanProgressProps) {
  // totalFiles = 0 during Pass 1 walk (indeterminate) — show pulse animation
  const isIndeterminate = !progress || progress.totalFiles === 0
  const percent = isIndeterminate
    ? 0
    : Math.round((progress.filesScanned / progress.totalFiles) * 100)

  const formatElapsed = (ms: number) => {
    const s = Math.floor(ms / 1000)
    const m = Math.floor(s / 60)
    return m > 0 ? `${m}:${String(s % 60).padStart(2, '0')}` : `0:${String(s).padStart(2, '0')}`
  }

  return (
    <div className="flex flex-col items-center justify-center h-full">
      <div className="w-full max-w-[480px] flex flex-col gap-6 px-8">
        {/* Folder path */}
        <p className="text-xs text-muted-foreground truncate" title={folderPath}>
          {folderPath}
        </p>

        {/* Progress bar — indeterminate uses animate-pulse gradient, determinate fills blue-500 */}
        {isIndeterminate ? (
          <div className="h-2 w-full rounded-full bg-muted overflow-hidden">
            <div className="h-full w-full animate-pulse bg-gradient-to-r from-blue-500/40 via-blue-500 to-blue-500/40 bg-[length:200%_100%]" />
          </div>
        ) : (
          <Progress
            value={percent}
            className="h-2"
            aria-label="Scan progress"
          />
        )}

        {/* Stats */}
        <div className="flex flex-col gap-1 text-sm text-foreground">
          {progress ? (
            <>
              <span>Files scanned: {progress.filesScanned.toLocaleString()}{progress.totalFiles > 0 ? ` / ${progress.totalFiles.toLocaleString()}` : ''}</span>
              <span>Elapsed: {formatElapsed(progress.elapsedMs)}</span>
              <span className={progress.duplicatesFound > 0 ? 'text-blue-500' : ''}>
                Duplicates found: {progress.duplicatesFound}
              </span>
            </>
          ) : (
            <span className="text-muted-foreground">Preparing scan…</span>
          )}
        </div>

        {/* Current file */}
        {progress?.currentPath && (
          <p className="text-xs text-muted-foreground truncate" dir="ltr" title={progress.currentPath}>
            {progress.currentPath}
          </p>
        )}

        {/* Cancel button */}
        <Button
          variant="outline"
          className="w-full"
          onClick={onCancel}
          aria-label="Cancel current scan"
        >
          Cancel Scan
        </Button>
      </div>
    </div>
  )
}

export default ScanProgress
```

**DuplicatesPage.tsx** — Three-state orchestrator. Replace the current placeholder entirely:
```tsx
// src/renderer/src/pages/DuplicatesPage.tsx
import { useEffect } from 'react'
import { useAppStore } from '@/store/appStore'
import ScanEmpty from '@/components/ScanEmpty'
import ScanProgress from '@/components/ScanProgress'
import DuplicateGroups from '@/components/DuplicateGroups'
import { Button } from '@/components/ui/button'

function DuplicatesPage() {
  const selectedFolder = useAppStore((s) => s.selectedFolder)
  const scanStatus = useAppStore((s) => s.scanStatus)
  const scanProgress = useAppStore((s) => s.scanProgress)
  const duplicateGroups = useAppStore((s) => s.duplicateGroups)
  const { setScanProgress, setScanComplete, setScanError, setScanStatus, clearSelections, resetScan } = useAppStore()

  // Register IPC push-event listeners on mount — cleanup on unmount (Pattern 7, Pitfall 3)
  useEffect(() => {
    const unsubProgress = window.api.onScanProgress((data) => {
      setScanProgress(data)
    })
    const unsubComplete = window.api.onScanComplete((groups) => {
      setScanComplete(groups)
    })
    const unsubError = window.api.onScanError((msg) => {
      setScanError(msg)
    })

    return () => {
      unsubProgress()
      unsubComplete()
      unsubError()
    }
  }, [])  // Mount once — handlers read latest Zustand state, not stale closures

  const handleScanStart = async () => {
    if (!selectedFolder) return
    clearSelections()
    resetScan()
    setScanStatus('scanning')
    try {
      await window.api.scanStart(selectedFolder)
    } catch (err) {
      setScanError((err as Error).message ?? 'Failed to start scan')
    }
  }

  const handleScanCancel = async () => {
    setScanStatus('cancelled')
    await window.api.scanCancel()
  }

  // Derive which state to show
  const isScanning = scanStatus === 'scanning'
  const hasResults = scanStatus === 'complete' && duplicateGroups.length > 0
  const hasNoResults = scanStatus === 'complete' && duplicateGroups.length === 0
  const hasError = scanStatus === 'error'

  return (
    <div className="flex flex-col h-full relative">
      {/* Page header — always visible */}
      <div className="flex items-center justify-between px-6 py-4 border-b border-border shrink-0">
        <h1 className="text-xl font-semibold text-foreground">Duplicate Files</h1>
        {selectedFolder && !isScanning && (
          <Button
            className="bg-blue-500 hover:bg-blue-500/90 active:bg-blue-500/80"
            onClick={handleScanStart}
          >
            {hasResults || hasNoResults ? 'Scan Again' : 'Scan Now'}
          </Button>
        )}
      </div>

      {/* Content area */}
      <div className="flex-1 overflow-hidden">
        {isScanning && selectedFolder && (
          <ScanProgress
            folderPath={selectedFolder}
            progress={scanProgress}
            onCancel={handleScanCancel}
          />
        )}

        {hasResults && (
          <DuplicateGroups groups={duplicateGroups} />
        )}

        {(hasNoResults || (!isScanning && !hasResults) || hasError) && (
          <div className="flex flex-col h-full">
            {hasError && (
              <div className="mx-6 mt-4 p-3 rounded-md bg-destructive/10 text-sm text-destructive border border-destructive/20">
                {/* Error message from useAppStore scanError */}
                <ErrorMessage />
              </div>
            )}
            <ScanEmpty hasFolder={!!selectedFolder && (hasNoResults)} />
          </div>
        )}
      </div>
    </div>
  )
}

// Inline helper to read scanError from store
function ErrorMessage() {
  const scanError = useAppStore((s) => s.scanError)
  return <>{scanError ?? 'Scan failed. Check that the folder exists and you have read permission.'}</>
}

export default DuplicatesPage
```

```tsx
// src/renderer/src/components/FileRow.tsx
import { Checkbox } from '@/components/ui/checkbox'
import { Badge } from '@/components/ui/badge'
import { Tooltip, TooltipContent, TooltipProvider, TooltipTrigger } from '@/components/ui/tooltip'
import { cn, formatBytes } from '@/lib/utils'
import type { FileEntry } from '../../../shared/ipc-types'

interface FileRowProps {
  file: FileEntry
  isChecked: boolean
  isKeep: boolean
  isDisabled: boolean  // keep-at-least-one: true when this is the last unchecked copy
  onToggle: (path: string) => void
}

function FileRow({ file, isChecked, isKeep, isDisabled, onToggle }: FileRowProps) {
  const formattedDate = new Date(file.mtime).toLocaleDateString('en-US', {
    month: 'short', day: 'numeric', year: 'numeric'
  })

  const checkbox = (
    <Checkbox
      checked={isChecked}
      disabled={isDisabled}
      onCheckedChange={() => !isDisabled && onToggle(file.path)}
      aria-label={file.path}
      className={cn(
        'size-5',
        isChecked && 'data-[state=checked]:bg-blue-500 data-[state=checked]:border-blue-500'
      )}
    />
  )

  return (
    <div className={cn(
      'flex items-center gap-2 px-4 py-2 text-sm',
      isDisabled && 'opacity-50',
      !isDisabled && 'hover:bg-muted/50'
    )}>
      {/* Checkbox — wrap in tooltip if disabled */}
      {isDisabled ? (
        <TooltipProvider>
          <Tooltip>
            <TooltipTrigger asChild>
              <span>{checkbox}</span>
            </TooltipTrigger>
            <TooltipContent>Cannot remove last copy</TooltipContent>
          </Tooltip>
        </TooltipProvider>
      ) : (
        checkbox
      )}

      {/* Path — truncate with ellipsis */}
      <span
        className={cn('flex-1 truncate', isChecked ? 'text-foreground' : 'text-muted-foreground')}
        title={file.path}
      >
        {file.path}
      </span>

      {/* Size */}
      <span className="text-xs text-muted-foreground w-16 text-right shrink-0">
        {formatBytes(file.size)}
      </span>

      {/* Modified date */}
      <span className="text-xs text-muted-foreground w-24 text-right shrink-0">
        {formattedDate}
      </span>

      {/* KEEP badge or spacer — green semantic color per UI-SPEC */}
      <div className="w-12 flex justify-end shrink-0">
        {isKeep ? (
          <Badge
            className="bg-emerald-500/20 text-emerald-400 text-xs font-medium px-2 py-0.5 rounded-full border-0"
            aria-label="This copy will be kept"
          >
            KEEP
          </Badge>
        ) : (
          <span className="w-12" />
        )}
      </div>
    </div>
  )
}

export default FileRow
```

**DuplicateGroupCard.tsx** — Single accordion item wrapping file rows:
```tsx
// src/renderer/src/components/DuplicateGroupCard.tsx
import { AccordionContent, AccordionItem, AccordionTrigger } from '@/components/ui/accordion'
import FileRow from './FileRow'
import { formatBytes } from '@/lib/utils'
import { useAppStore } from '@/store/appStore'
import type { DuplicateGroup } from '../../../shared/ipc-types'

interface DuplicateGroupCardProps {
  group: DuplicateGroup
}

function DuplicateGroupCard({ group }: DuplicateGroupCardProps) {
  const selections = useAppStore((s) => s.selections)
  const toggleSelection = useAppStore((s) => s.toggleSelection)

  // Determine which file is the KEEP candidate: the one not checked for deletion
  // If no preset applied yet, no file has KEEP badge (isKeep = false for all)
  const checkedCount = group.files.filter((f) => selections.has(f.path)).length
  const uncheckedFiles = group.files.filter((f) => !selections.has(f.path))

  // keep-at-least-one: the last unchecked file in a group is disabled (SELECT-02)
  const isLastUnchecked = (filePath: string) =>
    uncheckedFiles.length === 1 && uncheckedFiles[0].path === filePath && !selections.has(filePath)

  const totalSize = group.files.reduce((sum, f) => sum + f.size, 0)
  const fileName = group.files[0]?.name ?? 'Unknown file'

  return (
    <AccordionItem
      value={group.hash}
      className="border border-border rounded-lg mb-2 overflow-hidden data-[state=open]:border-blue-500/30"
    >
      <AccordionTrigger className="px-4 py-3 hover:bg-muted/50 hover:no-underline">
        <div className="flex items-center gap-2 text-left">
          <span className="text-sm font-semibold text-foreground">{fileName}</span>
          <span className="text-xs text-muted-foreground">
            · {group.files.length} copies · {formatBytes(totalSize)}
          </span>
        </div>
      </AccordionTrigger>
      <AccordionContent className="pb-0">
        <div className="border-t border-border">
          {group.files.map((file) => {
            const isChecked = selections.has(file.path)
            // KEEP badge: show on unchecked files when at least one in the group is checked
            const isKeep = !isChecked && checkedCount > 0
            const isDisabled = isLastUnchecked(file.path)

            return (
              <FileRow
                key={file.path}
                file={file}
                isChecked={isChecked}
                isKeep={isKeep}
                isDisabled={isDisabled}
                onToggle={toggleSelection}
              />
            )
          })}
        </div>
      </AccordionContent>
    </AccordionItem>
  )
}

export default DuplicateGroupCard
```

```tsx
// src/renderer/src/components/DuplicateGroups.tsx
import { useState } from 'react'
import { Accordion } from '@/components/ui/accordion'
import { Button } from '@/components/ui/button'
import { ScrollArea } from '@/components/ui/scroll-area'
import DuplicateGroupCard from './DuplicateGroupCard'
import { useAppStore, applyPreset } from '@/store/appStore'
import { formatBytes } from '@/lib/utils'
import type { DuplicateGroup } from '../../../shared/ipc-types'

interface DuplicateGroupsProps {
  groups: DuplicateGroup[]
}

function DuplicateGroups({ groups }: DuplicateGroupsProps) {
  const protectedFolders = useAppStore((s) => s.protectedFolders)
  const setSelections = useAppStore((s) => s.setSelections)

  // D-09: First group open on results load; rest collapsed
  const [openItems, setOpenItems] = useState<string[]>(
    groups.length > 0 ? [groups[0].hash] : []
  )

  // Space savings: sum of totalWastedBytes across all groups (DUPE-04)
  const totalWastedBytes = groups.reduce((sum, g) => sum + g.totalWastedBytes, 0)
  const duplicateCount = groups.reduce((sum, g) => sum + g.files.length - 1, 0)

  const handlePreset = (preset: 'newest' | 'oldest') => {
    const newSelections = applyPreset(groups, preset, protectedFolders)
    setSelections(newSelections)
  }

  // Space savings summary copy (DUPE-04, Copywriting Contract)
  const summaryCopy = duplicateCount === 1
    ? `1 duplicate found — you can reclaim ${formatBytes(totalWastedBytes)}`
    : `${duplicateCount} duplicates found — you can reclaim ${formatBytes(totalWastedBytes)}`

  return (
    <div className="flex flex-col h-full">
      {/* Summary bar — sticky, does not scroll (D-10) */}
      <div
        className="sticky top-0 z-10 bg-background flex items-center justify-between px-6 py-3 border-b border-border shrink-0"
        aria-live="polite"
      >
        <span className="text-sm text-foreground">{summaryCopy}</span>
        <div className="flex items-center gap-2">
          <Button
            variant="outline"
            size="sm"
            className="border-blue-500 text-blue-500 hover:bg-blue-500/10"
            onClick={() => handlePreset('newest')}
          >
            Keep Newest
          </Button>
          <Button
            variant="outline"
            size="sm"
            className="border-blue-500 text-blue-500 hover:bg-blue-500/10"
            onClick={() => handlePreset('oldest')}
          >
            Keep Oldest
          </Button>
        </div>
      </div>

      {/* Scrollable accordion list — add pb-20 to clear the sticky DeletePanel (plan 02-05) */}
      <ScrollArea className="flex-1">
        <div className="px-6 py-4 pb-20">
          <Accordion
            type="multiple"
            value={openItems}
            onValueChange={setOpenItems}
          >
            {groups.map((group) => (
              <DuplicateGroupCard key={group.hash} group={group} />
            ))}
          </Accordion>
        </div>
      </ScrollArea>
    </div>
  )
}

export default DuplicateGroups
```
