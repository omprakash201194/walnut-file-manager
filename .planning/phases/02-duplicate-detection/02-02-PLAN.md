```typescript
import { BrowserWindow, ipcMain, shell } from 'electron'
import { promises as fs } from 'fs'
import * as crypto from 'crypto'
import * as path from 'path'
import { IPC_CHANNELS, FileEntry, DuplicateGroup, ScanProgressEvent, DeleteResult } from '../../../shared/ipc-types'

// Module-level controller — allows SCAN_CANCEL to abort an in-progress scan
let scanController: AbortController | null = null

// PASS 1: Iterative (not recursive) directory walk — avoids call stack overflow on deep trees
// Uses a stack array instead of recursion; emits onFile callback for each file found
// Skips entries on stat/readdir error (CLAUDE.md: always handle permission errors)
export async function walkDirectory(
  dirPath: string,
  signal: AbortSignal,
  onFile: (entry: FileEntry, count: number) => void
): Promise<FileEntry[]>

// DUPE-01: Group collected files by size; return only groups with 2+ files
// Files with unique sizes are not duplicate candidates and are discarded
export function groupBySize(files: FileEntry[]): Map<number, FileEntry[]>

// DUPE-02: Stream SHA-256 hash via createReadStream — non-blocking, handles large files
// Wraps in try/catch — returns null on permission error (Pitfall 4)
export async function hashFile(filePath: string): Promise<string | null>

// DUPE-02: Hash all candidates; group by hash; filter out singletons
export async function groupByHash(
  candidates: FileEntry[],
  signal: AbortSignal,
  onProgress: (hashed: number, total: number, currentPath: string) => void
): Promise<DuplicateGroup[]>

// Register all scan IPC handlers on the mainWindow
export function registerScanHandlers(mainWindow: BrowserWindow): void {
  ipcMain.handle(IPC_CHANNELS.SCAN_START, async (_, folderPath: string) => {
    // Abort any in-progress scan first
    scanController?.abort()
    scanController = new AbortController()
    const { signal } = scanController
    const startTime = Date.now()

    try {
      // PASS 1: Walk directory, emit progress every ~100 files
      const files: FileEntry[] = []
      await walkDirectory(folderPath, signal, (entry, count) => {
        files.push(entry)
        if (count % 100 === 0) {
          const progressEvent: ScanProgressEvent = {
            filesScanned: count,
            totalFiles: 0,          // Unknown during walk
            currentPath: entry.path,
            duplicatesFound: 0,
            elapsedMs: Date.now() - startTime
          }
          mainWindow.webContents.send(IPC_CHANNELS.SCAN_PROGRESS, progressEvent)
        }
      })

      // DUPE-01: Size pre-filter
      const sizeGroups = groupBySize(files)
      const candidates = [...sizeGroups.values()].flat()

      // PASS 2: Hash candidates, stream progress
      let hashCount = 0
      const duplicateGroups = await groupByHash(candidates, signal, (hashed, total, currentPath) => {
        hashCount = hashed
        const progressEvent: ScanProgressEvent = {
          filesScanned: files.length,
          totalFiles: total,
          currentPath,
          duplicatesFound: 0,    // Will be updated below but send running total
          elapsedMs: Date.now() - startTime
        }
        mainWindow.webContents.send(IPC_CHANNELS.SCAN_PROGRESS, progressEvent)
      })

      mainWindow.webContents.send(IPC_CHANNELS.SCAN_COMPLETE, duplicateGroups)
    } catch (err: unknown) {
      const isAbort = (err as { name?: string }).name === 'AbortError'
      if (!isAbort) {
        mainWindow.webContents.send(IPC_CHANNELS.SCAN_ERROR, (err as Error).message ?? 'Unknown scan error')
      }
      // AbortError = user cancelled — no SCAN_ERROR event
    } finally {
      scanController = null
    }
    // Returns void — CRITICAL: Do NOT return groups from handle (Pitfall 2)
  })

  ipcMain.handle(IPC_CHANNELS.SCAN_CANCEL, async () => {
    scanController?.abort()
  })

  ipcMain.handle(IPC_CHANNELS.DELETE_FILES, async (_, paths: string[]): Promise<DeleteResult> => {
    const succeeded: string[] = []
    const failed: Array<{ path: string; error: string }> = []
    for (const filePath of paths) {
      try {
        // path.resolve() converts forward slashes to backslashes on Windows (Pitfall 1)
        await shell.trashItem(path.resolve(filePath))
        succeeded.push(filePath)
      } catch (err) {
        failed.push({ path: filePath, error: (err as Error).message ?? 'Unknown error' })
      }
    }
    return { succeeded, failed }
  })
}
```

Implementation notes for walkDirectory:
- Use an iterative stack: `const stack: string[] = [dirPath]`
- Pop from stack, call `fs.readdir(dir, { withFileTypes: true })`
- For each dirEntry: if directory, push to stack; if file, call `fs.stat()` to get size and mtime
- Wrap `readdir` and `stat` calls in try/catch and skip on error (Pitfall 4)
- Check `signal.aborted` at the top of each loop iteration; if true, throw new DOMException('Aborted', 'AbortError')

Implementation notes for hashFile:
- `crypto.createHash('sha256')` piped from `fs.createReadStream(filePath)`
- Return hex digest string
- Wrap in try/catch; return null on any error (permission denied, locked file)

Implementation notes for groupByHash:
- Map<string, FileEntry[]> accumulating by hash
- For each candidate: await hashFile; skip if null; add to map
- After all: filter map entries with 2+ files
- Compute totalWastedBytes = entry.size * (files.length - 1)
- Check signal.aborted before each hashFile call
